cmake_minimum_required(VERSION 3.20)
project(obiwan_recomp C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "This bring-up launcher currently targets x86-64 Linux")
endif()
set(_runtime_default "${CMAKE_CURRENT_SOURCE_DIR}/../xbox-linux/third_party/xboxrecomp")
if(DEFINED ENV{XBOXRECOMP_DIR} AND NOT "$ENV{XBOXRECOMP_DIR}" STREQUAL "")
  set(_runtime_default "$ENV{XBOXRECOMP_DIR}")
endif()
set(XBOXRECOMP_DIR "${_runtime_default}" CACHE PATH "Path to separately cloned Xbox Linux runtime dependency")
find_package(OpenSSL REQUIRED)
add_subdirectory("${XBOXRECOMP_DIR}" "${CMAKE_BINARY_DIR}/runtime")
file(GLOB GENERATED CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen/*.c")
if(NOT GENERATED)
  message(FATAL_ERROR "Run tools/generate_obiwan.py before configuring this project")
endif()
add_executable(obiwan_recomp src/main.c src/manual.c ${GENERATED})
target_include_directories(obiwan_recomp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen")
target_link_libraries(obiwan_recomp PRIVATE xboxrecomp OpenSSL::Crypto)
# Start with debug-friendly code; guest semantics must not depend on host aliasing rules.
target_compile_options(obiwan_recomp PRIVATE -fno-strict-aliasing -fwrapv)
target_link_options(obiwan_recomp PRIVATE -rdynamic)
